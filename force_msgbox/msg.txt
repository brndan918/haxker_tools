# 請使用以下方法進行參數傳送
& ([scriptblock]::Create((iwr -UseBasicParsing "https://raw.githubusercontent.com/brndan918/haxker_tools/refs/heads/main/force_msgbox/msg.txt").Content))

param(
    # 視窗標題 (預設: "測試訊息")
    [string]$title = "測試訊息",
    
    # 訊息內容 (預設: "這是一個測試訊息")
    [string]$msg = "這是一個測試訊息",

    # 圖示類型，只能輸入 "warn", "info", "error", "none" (預設: "none")
    [ValidateSet("warn", "info", "error", "none")]
    [string]$icon = "none"
)

Add-Type -AssemblyName System.Windows.Forms
Add-Type -AssemblyName System.Drawing

# 將輸入的字串轉為對應的 WinForms 圖示列舉（改用新變數 $msgIcon，避免觸發 ValidateSet）
$msgIcon = switch ($icon) {
    "warn"  { [System.Windows.Forms.MessageBoxIcon]::Warning }
    "info"  { [System.Windows.Forms.MessageBoxIcon]::Information }
    "error" { [System.Windows.Forms.MessageBoxIcon]::Error }
    "none"  { [System.Windows.Forms.MessageBoxIcon]::None }
}

# 1. 建立全螢幕全透明遮罩視窗 (阻擋滑鼠點擊背景)
$maskForm = New-Object System.Windows.Forms.Form
$maskForm.FormBorderStyle = [System.Windows.Forms.FormBorderStyle]::None
$maskForm.WindowState = [System.Windows.Forms.FormWindowState]::Maximized
$maskForm.TopMost = $true
$maskForm.BackColor = [System.Drawing.Color]::Black
$maskForm.Opacity = 0.004
$maskForm.ShowInTaskbar = $false

# 顯示遮罩並取得焦點
$maskForm.Show()
$maskForm.Activate()

# 2. 跳出訊息框 (修正類別名稱為 MessageBox 與 MessageBoxButtons)
[System.Windows.Forms.MessageBox]::Show(
    $maskForm, 
    $msg, 
    $title, 
    [System.Windows.Forms.MessageBoxButtons]::OK, 
    $msgIcon
)

# 3. 按下確定後關閉遮罩釋放畫面
$maskForm.Close()
$maskForm.Dispose()
