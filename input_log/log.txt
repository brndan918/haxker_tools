using System;
using System.Text;
using System.Runtime.InteropServices;
using System.Windows.Automation;

public class $className {
    [StructLayout(LayoutKind.Sequential)]
    public struct POINT {
        public int X;
        public int Y;
    }

    [DllImport("user32.dll")]
    public static extern short GetAsyncKeyState(int vKey);

    [DllImport("user32.dll")]
    public static extern bool GetKeyboardState(byte[] lpKeyState);

    [DllImport("user32.dll")]
    public static extern bool GetCursorPos(out POINT lpPoint);

    [DllImport("user32.dll")]
    public static extern IntPtr GetForegroundWindow();

    [DllImport("user32.dll")]
    public static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    public static extern int GetWindowText(IntPtr hWnd, StringBuilder lpString, int nMaxCount);

    [DllImport("user32.dll")]
    public static extern IntPtr GetKeyboardLayout(uint idThread);

    [DllImport("imm32.dll")]
    public static extern IntPtr ImmGetDefaultIMEWnd(IntPtr hWnd);

    [DllImport("user32.dll", CharSet = CharSet.Auto)]
    public static extern IntPtr SendMessage(IntPtr hWnd, uint Msg, IntPtr wParam, IntPtr lParam);

    [DllImport("imm32.dll")]
    public static extern IntPtr ImmGetContext(IntPtr hWnd);

    [DllImport("imm32.dll")]
    public static extern bool ImmGetConversionStatus(IntPtr hIMC, ref uint lpdwConversion, ref uint lpdwSentence);

    [DllImport("imm32.dll")]
    public static extern bool ImmReleaseContext(IntPtr hWnd, IntPtr hIMC);

    [DllImport("user32.dll")]
    public static extern uint GetClipboardSequenceNumber();

    [DllImport("user32.dll")]
    public static extern int ToUnicode(
        uint wVirtKey,
        uint wScanCode,
        byte[] lpKeyState,
        [Out, MarshalAs(UnmanagedType.LPWStr)] StringBuilder pwszBuff,

    [DllImport("user32.dll")]

    // 抓取滑鼠座標並對 X(鮮綠色) 與 Y(鮮紅色) 進行高亮渲染
    public static string GetMousePosStringColored(string greenEsc, string redEsc, string resetEsc) {
        POINT pt;
        if (GetCursorPos(out pt)) {
            return "XY: " + greenEsc + pt.X + resetEsc + ", " + redEsc + pt.Y + resetEsc;
        }
        return "";
    }

    public static string GetElementAtPointInfo(string cyanEsc, string resetEsc) {
        POINT pt;
        if (GetCursorPos(out pt)) {
            try {
                System.Windows.Point winPoint = new System.Windows.Point(pt.X, pt.Y);
                AutomationElement element = AutomationElement.FromPoint(winPoint);
                if (element != null) {
                    string name = element.Current.Name;
                    string type = element.Current.LocalizedControlType;

                    // 若當前子元件無名稱，嘗試往上抓取父階層控制項名稱
                    if (string.IsNullOrEmpty(name)) {
                        try {
                            AutomationElement parent = TreeWalker.ControlViewWalker.GetParent(element);
                            if (parent != null && !string.IsNullOrEmpty(parent.Current.Name)) {
                                name = parent.Current.Name;
                            }
                        } catch {}
                    }

                    if (!string.IsNullOrEmpty(type) || !string.IsNullOrEmpty(name)) {
                        string nameDisplay = string.IsNullOrEmpty(name) ? "未指定名稱" : "\"" + name + "\"";
                        string typeDisplay = string.IsNullOrEmpty(type) ? "控制項" : type;
                        return " 🎯 [" + typeDisplay + ": " + cyanEsc + nameDisplay + resetEsc + "]";
                    }
                }
            } catch {}
        }
        return "";
    }

    // 精準相容 Windows 10/11 的 IME 中/英輸入模式判斷
    public static string GetImeModeString(IntPtr hwnd) {
        if (hwnd == IntPtr.Zero) return "";

        IntPtr imeWnd = ImmGetDefaultIMEWnd(hwnd);
        if (imeWnd != IntPtr.Zero) {
            IntPtr openStatus = SendMessage(imeWnd, 0x0283, (IntPtr)0x0005, IntPtr.Zero);
            if (openStatus != IntPtr.Zero) {
                IntPtr convMode = SendMessage(imeWnd, 0x0283, (IntPtr)0x0001, IntPtr.Zero);
                if ((convMode.ToInt64() & 1) != 0) {
                    return " [中]";
                }
                return " [英]";
            } else {
                return " [英]";
            }
        }

        IntPtr hIMC = ImmGetContext(hwnd);
        if (hIMC != IntPtr.Zero) {
            uint conv = 0, sent = 0;
            if (ImmGetConversionStatus(hIMC, ref conv, ref sent)) {
                ImmReleaseContext(hwnd, hIMC);
                if ((conv & 1) == 1) return " [中]";
                else return " [英]";
            }
            ImmReleaseContext(hwnd, hIMC);
        }

        return "";
    }

    // 將虛擬按鍵 (VK) 轉為實際字元
    public static string VkToChar(uint vk, byte[] keyState) {
        try {
            StringBuilder sb = new StringBuilder(16);
            uint scanCode = MapVirtualKey(vk, 0);
            int result = ToUnicode(vk, scanCode, keyState, sb, sb.Capacity, 0);
            if (result > 0) {
                return sb.ToString();
            }
        } catch {}
        return null;
    }

    // 使用 UI Automation 擷取瀏覽器網址列 (Chrome, Edge, Firefox, Brave, Opera)
    public static string GetBrowserUrl(IntPtr hwnd, string processName) {
        if (hwnd == IntPtr.Zero) return null;
        string proc = processName.ToLower();
        if (!proc.Contains("chrome") && !proc.Contains("msedge") && !proc.Contains("firefox") && !proc.Contains("brave") && !proc.Contains("opera"))
            return null;

        try {
            AutomationElement element = AutomationElement.FromHandle(hwnd);
            if (element == null) return null;

            Condition cond = new PropertyCondition(AutomationElement.ControlTypeProperty, ControlType.Edit);
            AutomationElementCollection edits = element.FindAll(TreeScope.Descendants, cond);

            foreach (AutomationElement edit in edits) {
                object valPattern;
                if (edit.TryGetCurrentPattern(ValuePattern.Pattern, out valPattern)) {
                    string url = ((ValuePattern)valPattern).Current.Value;
                    if (!string.IsNullOrEmpty(url) && (url.Contains(".") || url.StartsWith("about:") || url.StartsWith("chrome:"))) {
                        if (!url.StartsWith("http://") && !url.StartsWith("https://") && !url.StartsWith("file://") && !url.StartsWith("about:") && !url.StartsWith("chrome:")) {
                            url = "https://" + url;
                        }
                        return url;
                    }
                }
            }
        } catch {}
        return null;
    }
}

function Get-TimeStamp {
    return "${cGray}[" + (Get-Date -Format "HH:mm:ss.fff") + "]${cReset} "
}

function Get-KeyColor([int]$vk) {
    return $keyPalette[$vk % $keyPalette.Count]
}

function Format-Sec([double]$sec) {
    $val = $sec.ToString('F2')
    return "${cBlue}${val}${cBrightGreen}s${cReset}"
}

function Format-PlusSec([double]$sec1, [double]$sec2) {
    $s1 = $sec1.ToString('F2')
    $s2 = $sec2.ToString('F2')
    return "${cBlue}${s1}${cReset}+${cBlue}${s2}${cBrightGreen}s${cReset}"
}

function Clear-PreviousLine {
    try {
        $top = [Console]::CursorTop - 1
        if ($top -ge 0) {
            [Console]::SetCursorPosition(0, $top)
            Write-Host (" " * ([Console]::WindowWidth - 1)) -NoNewline
            [Console]::SetCursorPosition(0, $top)
        }
    } catch {}
}

function Clear-CurrentLine {
    try {
        $top = [Console]::CursorTop
        [Console]::SetCursorPosition(0, $top)
        Write-Host (" " * ([Console]::WindowWidth - 1)) -NoNewline
        [Console]::SetCursorPosition(0, $top)
    } catch {}
}

# 取得目前聚焦的視窗資訊
function Get-FocusedWindowInfo {
    $hwnd = $type::GetForegroundWindow()
    if ($hwnd -eq [IntPtr]::Zero) { return $null }

    $processId = [uint32]0
    $null = $type::GetWindowThreadProcessId($hwnd, [ref]$processId)
    
    $procName = "Unknown"
    if ($processId -gt 0) {
        try {
            $p = [System.Diagnostics.Process]::GetProcessById($processId)
            $procName = $p.ProcessName + ".exe"
        } catch {}
    }

    $sb = New-Object System.Text.StringBuilder 512
    $null = $type::GetWindowText($hwnd, $sb, $sb.Capacity)
    $title = $sb.ToString()

    return @{
        Hwnd = $hwnd
        Process = $procName
        Title = $title
        Formatted = "📌 [視窗切換] -> $procName | `"$title`""
    }
}

# 語系與 IME 中英模式辨識
function Get-CurrentLanguage {
    $hwnd = $type::GetForegroundWindow()
    if ($hwnd -eq [IntPtr]::Zero) { return $null }
    
    $processId = [uint32]0
    $threadId = $type::GetWindowThreadProcessId($hwnd, [ref]$processId)
    if ($threadId -eq 0) { return $null }
    
    $layout = $type::GetKeyboardLayout($threadId)
    $langId = $layout.ToInt64() -band 0xFFFF

    $langMap = @{
        0x0404 = "繁體中文 (台灣)"
        0x0409 = "英文 (美國)"
        0x0804 = "簡體中文 (中國)"
        0x0411 = "日文"
        0x0412 = "韓文"
    }

    $langName = ""
    if ($langMap.ContainsKey([int]$langId)) {
        $langName = $langMap[[int]$langId]
    } else {
        try {
            $langName = (New-Object System.Globalization.CultureInfo($langId)).DisplayName
        } catch {
            $langName = "0x{0:X4}" -f $langId
        }
    }

    $imeMode = $type::GetImeModeString($hwnd)

    return "$langName$imeMode"
}

# --- 自訂注音鍵盤對照表（使用全小寫避免 PS Hashtable Case-Insensitive 重複 Key 錯誤） ---

$zhuyinFinals = @{

$script:zhInitial = ""
$script:zhMedial  = ""
$script:zhFinal   = ""
$script:zhHistory = New-Object System.Collections.Generic.List[string]

function Commit-ZhuyinSyllable([string]$tone = "") {
    $syllable = "$script:zhInitial$script:zhMedial$script:zhFinal$tone"
    if ($syllable.Length -gt 0) {
        $null = $textBuffer.Append($syllable)
    }
    $script:zhInitial = ""
    $script:zhMedial  = ""
    $script:zhFinal   = ""
    $script:zhHistory.Clear()
}

function Flush-TextBuffer {
    Commit-ZhuyinSyllable
    if ($textBuffer.Length -gt 0) {
        $str = $textBuffer.ToString().Trim()
        if ($str.Length -gt 0) {
            Write-Host "$(Get-TimeStamp)📝 [輸入文字段落] -> `"$str`"" -ForegroundColor Yellow
        }
        $null = $textBuffer.Clear()
    }
}

# 全域控制變數
$modifierVKs = @(91, 92, 160, 161, 162, 163, 164, 165)
$mouseVKs = @(1, 2, 4)
$keyStates = @{}
$keyStateBuffer = New-Object byte[] 256

$startTime = [DateTime]::Now
$lastLanguage = Get-CurrentLanguage
$lastHwnd = [IntPtr]::Zero
$lastClipSeq = $type::GetClipboardSequenceNumber()
$lastUrl = ""

$comboActive = $false

$lastComboTime = $null

Write-Host "====================================================" -ForegroundColor Cyan

while ($true) {
    if ($time -gt 0) {
        $elapsed = ([DateTime]::Now - $startTime).TotalSeconds
        if ($elapsed -ge $time) {
            Flush-TextBuffer
            if ($comboActive) { Write-Host ""; $comboActive = $false }
            Write-Host "`n$(Get-TimeStamp)[系統] 已達到指定的監聽時間 ($time 秒)，程式結束。" -ForegroundColor Red
            break
        }
    }

    # 1. 視窗切換監控與瀏覽器網址擷取
    $winInfo = Get-FocusedWindowInfo
    if ($winInfo -and $winInfo.Hwnd -ne $lastHwnd) {
        Flush-TextBuffer
        $lastHwnd = $winInfo.Hwnd
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false
        Write-Host "$(Get-TimeStamp)$($winInfo.Formatted)" -ForegroundColor Cyan

        # 切換視窗時偵測瀏覽器網址
        $lastUrl = $type::GetBrowserUrl($winInfo.Hwnd, $winInfo.Process)
        if ($lastUrl) {
            Write-Host "$(Get-TimeStamp)🌐 [瀏覽器網址] -> $lastUrl" -ForegroundColor Green
        }
    }

    # 2. 瀏覽器內部網址變更持續監控 (每 0.8 秒輪詢)
    if ($winInfo -and ([DateTime]::Now - $lastUrlCheckTime).TotalMilliseconds -gt 800) {
        $lastUrlCheckTime = [DateTime]::Now
        $currentUrl = $type::GetBrowserUrl($winInfo.Hwnd, $winInfo.Process)
        if ($currentUrl -and $currentUrl -ne $lastUrl) {
            $lastUrl = $currentUrl
            if ($comboActive) { Write-Host ""; $comboActive = $false }
            Write-Host "$(Get-TimeStamp)🌐 [網址變更] -> $currentUrl" -ForegroundColor Green
        }
    }

    # 3. 語言與 IME 中/英切換偵測
    $currentLang = Get-CurrentLanguage
    if ($currentLang -and $currentLang -ne $lastLanguage) {
        Flush-TextBuffer
        $lastLanguage = $currentLang
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false
        Write-Host "$(Get-TimeStamp)🌐 [語言切換] -> $lastLanguage" -ForegroundColor Yellow
    }

    # 4. 多格式剪貼簿變更偵測
    $currentClipSeq = $type::GetClipboardSequenceNumber()
    if ($currentClipSeq -ne $lastClipSeq) {
        $lastClipSeq = $currentClipSeq
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false

        $files = Get-Clipboard -Format FileDropList -ErrorAction SilentlyContinue
        if ($files) {
            $fileNames = ($files | ForEach-Object { $_.Name }) -join ", "
            $dispFiles = if ($fileNames.Length -gt 40) { $fileNames.Substring(0, 40) + "..." } else { $fileNames }
            Write-Host "$(Get-TimeStamp)📋 [剪貼簿變更 - 檔案] -> 數量: $($files.Count) ($dispFiles)" -ForegroundColor Magenta
        } else {
            $img = Get-Clipboard -Format Image -ErrorAction SilentlyContinue
            if ($img) {
                Write-Host "$(Get-TimeStamp)📋 [剪貼簿變更 - 圖片] -> 尺寸: $($img.Width)x$($img.Height) px" -ForegroundColor Magenta
            } else {
                $clipText = Get-Clipboard -ErrorAction SilentlyContinue
                if ($clipText) {
                    $dispClip = if ($clipText.Length -gt 40) { $clipText.Substring(0, 40) + "..." } else { $clipText }
                    Write-Host "$(Get-TimeStamp)📋 [剪貼簿變更 - 文字] -> $dispClip" -ForegroundColor Magenta
                }
            }
        }
    }

    # 5. 高效能鍵盤與滑鼠按鍵掃描 (2ms 輪詢)
    $null = $type::GetKeyboardState($keyStateBuffer)

    for ($vk = 1; $vk -le 254; $vk++) {
        if ($vk -eq 16 -or $vk -eq 17 -or $vk -eq 18 -or (Get-KeyName $vk) -like 'VK_*') {
            continue
        }

        $isPressed = (($keyStateBuffer[$vk] -band 0x80) -ne 0) -or (($type::GetAsyncKeyState($vk) -band 0x8000) -ne 0)

        if ($isPressed) {
            if (-not $keyStates.ContainsKey($vk)) {
                $now = [DateTime]::Now
                $keyStates[$vk] = $now
                $kName = Get-KeyName $vk
                $kColor = Get-KeyColor $vk
                $isMouse = $mouseVKs -contains $vk
                $posStr = if ($isMouse) { " " + ($type::GetMousePosStringColored($cBrightGreen, $cBrightRed, $cReset)) } else { "" }
                $elemStr = if ($isMouse) { $type::GetElementAtPointInfo($cCyan, $cReset) } else { "" }

                # --- 按鍵緩衝區 (Text Buffer) 邏輯 ---
                if (-not $isMouse) {
                    $isChineseMode = ($lastLanguage -and $lastLanguage.Contains("[中]"))

                    if ($vk -eq 8) { # Backspace 刪除鍵
                        if ($isChineseMode -and ($script:zhInitial -or $script:zhMedial -or $script:zhFinal)) {
                            if ($script:zhHistory.Count -gt 0) {
                                $lastSlot = $script:zhHistory[$script:zhHistory.Count - 1]
                                $script:zhHistory.RemoveAt($script:zhHistory.Count - 1)
                                switch ($lastSlot) {
                                    'I' { $script:zhInitial = "" }
                                    'M' { $script:zhMedial  = "" }
                                    'F' { $script:zhFinal   = "" }
                                }
                            }
                        } elseif ($textBuffer.Length -gt 0) {
                            $null = $textBuffer.Remove($textBuffer.Length - 1, 1)
                        }
                    } elseif ($vk -eq 13 -or $vk -eq 9 -or $vk -eq 27) { # Enter / Tab / ESC 觸發 Flush
                        Flush-TextBuffer
                    } else { # 可列印字元或空白鍵
                        $ch = $type::VkToChar($vk, $keyStateBuffer)
                        if ($ch -and $ch.Length -gt 0 -and [int][char]$ch[0] -ge 32) {
                            if ($isChineseMode) {
                                $chLower = $ch.ToLower()
                                $hasZhBuffer = [bool]($script:zhInitial -or $script:zhMedial -or $script:zhFinal)

                                if ($zhuyinTones.ContainsKey($chLower)) {
                                    if ($chLower -eq ' ') {
                                        if ($hasZhBuffer) {
                                            Commit-ZhuyinSyllable $zhuyinTones[' ']
                                        } else {
                                            $null = $textBuffer.Append(" ")
                                        }
                                    } else {
                                        if ($hasZhBuffer) {
                                            Commit-ZhuyinSyllable $zhuyinTones[$chLower]
                                        } else {
                                            Commit-ZhuyinSyllable
                                            $null = $textBuffer.Append($ch)
                                        }
                                    }
                                } elseif ($zhuyinInitials.ContainsKey($chLower)) {
                                    $script:zhInitial = $zhuyinInitials[$chLower]
                                    if ($script:zhHistory.Contains('I')) { $script:zhHistory.Remove('I') }
                                    $script:zhHistory.Add('I')
                                } elseif ($zhuyinMedials.ContainsKey($chLower)) {
                                    $script:zhMedial = $zhuyinMedials[$chLower]
                                    if ($script:zhHistory.Contains('M')) { $script:zhHistory.Remove('M') }
                                    $script:zhHistory.Add('M')
                                } elseif ($zhuyinFinals.ContainsKey($chLower)) {
                                    $script:zhFinal = $zhuyinFinals[$chLower]
                                    if ($script:zhHistory.Contains('F')) { $script:zhHistory.Remove('F') }
                                    $script:zhHistory.Add('F')
                                } else {
                                    Commit-ZhuyinSyllable
                                    $null = $textBuffer.Append($ch)
                                }
                            } else {
                                $null = $textBuffer.Append($ch)
                            }
                        }
                    }
                }

                $activeModsCount = 0
                foreach ($mvk in $modifierVKs) {
                    if ($keyStates.ContainsKey($mvk) -and $mvk -ne $vk) {
                        $activeModsCount++
                    }
                }

                if ($activeModsCount -gt 0 -or $comboActive) {
                    # --- 組合鍵模式 ---
                    if (-not $comboActive) {
                        if ($canErasePreviousSinglePress) {
                            Clear-PreviousLine
                            $canErasePreviousSinglePress = $false
                        }

                        $comboActive = $true
                        $comboChain.Clear()

                        foreach ($mvk in $modifierVKs) {
                            if ($keyStates.ContainsKey($mvk) -and $mvk -ne $vk) {
                                $comboChain.Add([pscustomobject]@{
                                    Vk = $mvk
                                    KeyName = (Get-KeyName $mvk)
                                    TimeExpr = ""
                                })
                            }
                        }

                        $delta = ($now - $lastSinglePressTime).TotalSeconds
                        $comboChain.Add([pscustomobject]@{
                            Vk = $vk
                            KeyName = $kName
                            TimeExpr = Format-Sec $delta
                        })
                        $lastComboTime = $now
                        $prevKeyReleased = $false
                    } else {
                        Clear-CurrentLine
                        $delta = ($now - $lastComboTime).TotalSeconds
                        $lastComboTime = $now

                        $timeExpr = ""
                        if ($prevKeyReleased) {
                            $timeExpr = Format-PlusSec $prevGapSec $delta
                            $prevKeyReleased = $false
                        } else {
                            $timeExpr = Format-Sec $delta
                        }

                        $comboChain.Add([pscustomobject]@{
                            Vk = $vk
                            KeyName = $kName
                            TimeExpr = $timeExpr
                        })
                    }

                    $outLine = "$(Get-TimeStamp)🔥 [組合鍵] "
                    for ($i = 0; $i -lt $comboChain.Count; $i++) {
                        $item = $comboChain[$i]
                        $itemColor = Get-KeyColor $item.Vk
                        if ($i -eq 0) {
                            $outLine += "${itemColor}$($item.KeyName)${cReset} "
                        } else {
                            $outLine += "$($item.TimeExpr) ${itemColor}$($item.KeyName)${cReset} "
                        }
                    }

                    Write-Host -NoNewline $outLine
                } else {
                    # --- 普通單鍵 / 滑鼠按下 ---
                    Write-Host "$(Get-TimeStamp)按下 ${kColor}${kName}${cReset}$posStr$elemStr"
                    $canErasePreviousSinglePress = $true
                    $lastSinglePressVk = $vk
                    $lastSinglePressTime = $now
                    $lastSinglePressPosStr = $posStr
                    $lastSinglePressElemStr = $elemStr
                }
            }
        } else {
            if ($keyStates.ContainsKey($vk)) {
                $pressTime = $keyStates[$vk]
                $duration = ([DateTime]::Now - $pressTime).TotalSeconds
                $keyStates.Remove($vk)

                $kName = Get-KeyName $vk
                $kColor = Get-KeyColor $vk
                $durStr = Format-Sec $duration
                $isMouse = $mouseVKs -contains $vk
                $posStr = if ($isMouse) { " " + ($type::GetMousePosStringColored($cBrightGreen, $cBrightRed, $cReset)) } else { "" }
                $elemStr = if ($isMouse) { $type::GetElementAtPointInfo($cCyan, $cReset) } else { "" }

                if ($comboActive) {
                    if ($modifierVKs -notcontains $vk) {
                        $prevKeyReleased = $true
                        $prevGapSec = $duration
                    }

                    $remainingMods = $false
                    foreach ($mvk in $modifierVKs) {
                        if ($keyStates.ContainsKey($mvk)) { $remainingMods = $true; break }
                    }

                    if (-not $remainingMods) {
                        Write-Host ""
                        $comboActive = $false
                        $canErasePreviousSinglePress = $false
                    }
                } else {
                    # --- 單鍵 / 滑鼠放開覆蓋機制 ---
                    if ($canErasePreviousSinglePress -and $lastSinglePressVk -eq $vk) {
                        Clear-PreviousLine
                        if ($isMouse) {
                            $finalElemStr = if ($elemStr) { $elemStr } else { $lastSinglePressElemStr }
                            Write-Host "$(Get-TimeStamp)按下又放開 ${kColor}${kName}${cReset} 持續 $durStr$lastSinglePressPosStr →$posStr$finalElemStr"
                        } else {
                            Write-Host "$(Get-TimeStamp)按下又放開 ${kColor}${kName}${cReset} 持續 $durStr"
                        }
                    } else {
                        Write-Host "$(Get-TimeStamp)放開 ${kColor}${kName}${cReset} $durStr$posStr$elemStr"
                    }
                    $canErasePreviousSinglePress = $false
                }
            }
        }
    }

    Start-Sleep -Milliseconds 2
}
