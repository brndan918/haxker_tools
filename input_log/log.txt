param(
    [Parameter(Mandatory=$true)]
    [int]$time
)

# 使用動態類別名稱，解決 Add-Type 類別名稱衝突導致 $type 變成 $null 的問題
$rand = Get-Random
$className = "WinAPIEx_$rand"

$code = @"
using System;
using System.Text;
using System.Runtime.InteropServices;
using System.Diagnostics;

public class $className {
    [StructLayout(LayoutKind.Sequential)]
    public struct POINT {
        public int X;
        public int Y;
    }

    [DllImport("user32.dll")]
    public static extern short GetAsyncKeyState(int vKey);

    [DllImport("user32.dll")]
    public static extern bool GetKeyboardState(byte[] lpKeyState);

    [DllImport("user32.dll")]
    public static extern bool GetCursorPos(out POINT lpPoint);

    [DllImport("user32.dll")]
    public static extern IntPtr GetForegroundWindow();

    [DllImport("user32.dll")]
    public static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    public static extern int GetWindowText(IntPtr hWnd, StringBuilder lpString, int nMaxCount);

    [DllImport("user32.dll")]
    public static extern IntPtr GetKeyboardLayout(uint idThread);

    [DllImport("imm32.dll")]
    public static extern IntPtr ImmGetContext(IntPtr hWnd);

    [DllImport("imm32.dll")]
    public static extern bool ImmGetConversionStatus(IntPtr hIMC, ref uint lpdwConversion, ref uint lpdwSentence);

    [DllImport("imm32.dll")]
    public static extern bool ImmReleaseContext(IntPtr hWnd, IntPtr hIMC);

    [DllImport("user32.dll")]
    public static extern uint GetClipboardSequenceNumber();

    // 原生 C# 滑鼠座標抓取 Helper (高效且不會丟出 Null 異常)
    public static string GetMousePosString() {
        POINT pt;
        if (GetCursorPos(out pt)) {
            return "XY: " + pt.X + ", " + pt.Y;
        }
        return "";
    }
}
"@

# 正確載入 Type
$type = Add-Type -TypeDefinition $code -PassThru

# ANSI 色彩定義
$ESC = [char]27
$cReset = "$ESC[0m"
$cBlue  = "$ESC[94m"  # 數字：藍色
$cGreen = "$ESC[32m"  # 秒數單位 s：綠色

# 按鍵色彩調色盤
$keyPalette = @(
    "$ESC[91m", # 鮮紅
    "$ESC[93m", # 鮮黃
    "$ESC[95m", # 洋紅
    "$ESC[96m", # 青色
    "$ESC[97m", # 白色
    "$ESC[33m", # 深黃
    "$ESC[35m"  # 深紫
)

function Get-KeyColor([int]$vk) {
    return $keyPalette[$vk % $keyPalette.Count]
}

function Format-Sec([double]$sec) {
    $val = $sec.ToString('F2')
    return "${cBlue}${val}${cGreen}s${cReset}"
}

function Format-PlusSec([double]$sec1, [double]$sec2) {
    $s1 = $sec1.ToString('F2')
    $s2 = $sec2.ToString('F2')
    return "${cBlue}${s1}${cReset}+${cBlue}${s2}${cGreen}s${cReset}"
}

# 擦除上一行/當前行 (供動態覆蓋使用)
function Clear-PreviousLine {
    try {
        $top = [Console]::CursorTop - 1
        if ($top -ge 0) {
            [Console]::SetCursorPosition(0, $top)
            Write-Host (" " * ([Console]::WindowWidth - 1)) -NoNewline
            [Console]::SetCursorPosition(0, $top)
        }
    } catch {}
}

function Clear-CurrentLine {
    try {
        $top = [Console]::CursorTop
        [Console]::SetCursorPosition(0, $top)
        Write-Host (" " * ([Console]::WindowWidth - 1)) -NoNewline
        [Console]::SetCursorPosition(0, $top)
    } catch {}
}

# 取得目前聚焦的視窗資訊
function Get-FocusedWindowInfo {
    $hwnd = $type::GetForegroundWindow()
    if ($hwnd -eq [IntPtr]::Zero) { return $null }

    $processId = [uint32]0
    $null = $type::GetWindowThreadProcessId($hwnd, [ref]$processId)
    
    $procName = "Unknown"
    if ($processId -gt 0) {
        try {
            $p = [System.Diagnostics.Process]::GetProcessById($processId)
            $procName = $p.ProcessName + ".exe"
        } catch {}
    }

    $sb = New-Object System.Text.StringBuilder 512
    $null = $type::GetWindowText($hwnd, $sb, $sb.Capacity)
    $title = $sb.ToString()

    return @{
        Hwnd = $hwnd
        Process = $procName
        Title = $title
        Formatted = "📌 [視窗切換] -> $procName | `"$title`""
    }
}

function Get-KeyName([int]$vk) {
    switch ($vk) {
        1   { return '左鍵' }
        2   { return '右鍵' }
        4   { return '中鍵' }
        8   { return 'Backspace' }
        9   { return 'Tab' }
        13  { return 'Enter' }
        20  { return 'Caps Lock' }
        27  { return 'ESC' }
        32  { return 'Space' }
        33  { return 'Page Up' }
        34  { return 'Page Down' }
        35  { return 'End' }
        36  { return 'Home' }
        37  { return '← (Left)' }
        38  { return '↑ (Up)' }
        39  { return '→ (Right)' }
        40  { return '↓ (Down)' }
        44  { return 'Print Screen' }
        45  { return 'Insert' }
        46  { return 'Delete' }
        91  { return 'LWin' }
        92  { return 'RWin' }
        93  { return '選單鍵' }
        144 { return 'Num Lock' }
        145 { return 'Scroll Lock' }
        160 { return 'LShift' }
        161 { return 'RShift' }
        162 { return 'LCtrl' }
        163 { return 'RCtrl' }
        164 { return 'LAlt' }
        165 { return 'RAlt' }
        default {
            if ($vk -ge 48 -and $vk -le 57) { return [char]$vk }
            if ($vk -ge 65 -and $vk -le 90) { return [char]$vk }
            if ($vk -ge 96 -and $vk -le 105) { return 'NumPad ' + ($vk - 96) }
            if ($vk -ge 112 -and $vk -le 123) { return 'F' + ($vk - 111) }
            switch ($vk) {
                186 { return ';' }
                187 { return '=' }
                188 { return ',' }
                189 { return '-' }
                190 { return '.' }
                191 { return '/' }
                192 { return '`' }
                219 { return '[' }
                220 { return '\' }
                221 { return ']' }
                222 { return "'" }
                default { return 'VK_' + $vk }
            }
        }
    }
}

# 語系與 IME 中英模式辨識
function Get-CurrentLanguage {
    $hwnd = $type::GetForegroundWindow()
    if ($hwnd -eq [IntPtr]::Zero) { return $null }
    
    $processId = [uint32]0
    $threadId = $type::GetWindowThreadProcessId($hwnd, [ref]$processId)
    if ($threadId -eq 0) { return $null }
    
    $layout = $type::GetKeyboardLayout($threadId)
    $langId = $layout.ToInt64() -band 0xFFFF

    $langMap = @{
        0x0404 = "繁體中文 (台灣)"
        0x0409 = "英文 (美國)"
        0x0804 = "簡體中文 (中國)"
        0x0411 = "日文"
        0x0412 = "韓文"
    }

    $langName = ""
    if ($langMap.ContainsKey([int]$langId)) {
        $langName = $langMap[[int]$langId]
    } else {
        try {
            $langName = (New-Object System.Globalization.CultureInfo($langId)).DisplayName
        } catch {
            $langName = "0x{0:X4}" -f $langId
        }
    }

    $imeMode = ""
    if ($langId -eq 0x0404 -or $langId -eq 0x0804) {
        $hIMC = $type::ImmGetContext($hwnd)
        if ($hIMC -ne [IntPtr]::Zero) {
            $conv = [uint32]0
            $sent = [uint32]0
            if ($type::ImmGetConversionStatus($hIMC, [ref]$conv, [ref]$sent)) {
                if (($conv -band 1) -eq 1) {
                    $imeMode = " [中]"
                } else {
                    $imeMode = " [英]"
                }
            }
            $null = $type::ImmReleaseContext($hwnd, $hIMC)
        }
    }

    return "$langName$imeMode"
}

# 全域狀態控制變數
$modifierVKs = @(91, 92, 160, 161, 162, 163, 164, 165)
$mouseVKs = @(1, 2, 4)
$keyStates = @{}
$keyStateBuffer = New-Object byte[] 256

$startTime = [DateTime]::Now
$lastLanguage = Get-CurrentLanguage
$lastHwnd = [IntPtr]::Zero
$lastClipSeq = $type::GetClipboardSequenceNumber()

$comboActive = $false
$comboChain = [System.Collections.Generic.List[psobject]]::new()
$lastComboTime = $null
$prevKeyReleased = $false
$prevGapSec = 0.0

$canErasePreviousSinglePress = $false
$lastSinglePressVk = 0
$lastSinglePressTime = $null

Write-Host "====================================================" -ForegroundColor Cyan
Write-Host "  資安測試終端：鍵盤/滑鼠/組合鍵/剪貼簿/視窗監控" -ForegroundColor Cyan
Write-Host "====================================================" -ForegroundColor Cyan
Write-Host ""
if ($lastLanguage) {
    Write-Host "🌐 [初始輸入語言] -> $lastLanguage" -ForegroundColor Yellow
}

while ($true) {
    if ($time -gt 0) {
        $elapsed = ([DateTime]::Now - $startTime).TotalSeconds
        if ($elapsed -ge $time) {
            if ($comboActive) { Write-Host ""; $comboActive = $false }
            Write-Host "`n[系統] 已達到指定的監聽時間 ($time 秒)，程式結束。" -ForegroundColor Red
            break
        }
    }

    # 1. 視窗切換監控
    $winInfo = Get-FocusedWindowInfo
    if ($winInfo -and $winInfo.Hwnd -ne $lastHwnd) {
        $lastHwnd = $winInfo.Hwnd
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false
        Write-Host $winInfo.Formatted -ForegroundColor Cyan
    }

    # 2. 語言與 IME 中/英切換偵測
    $currentLang = Get-CurrentLanguage
    if ($currentLang -and $currentLang -ne $lastLanguage) {
        $lastLanguage = $currentLang
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false
        Write-Host "🌐 [語言切換] -> $lastLanguage" -ForegroundColor Yellow
    }

    # 3. 多格式剪貼簿變更偵測
    $currentClipSeq = $type::GetClipboardSequenceNumber()
    if ($currentClipSeq -ne $lastClipSeq) {
        $lastClipSeq = $currentClipSeq
        if ($comboActive) { Write-Host ""; $comboActive = $false }
        $canErasePreviousSinglePress = $false

        $files = Get-Clipboard -Format FileDropList -ErrorAction SilentlyContinue
        if ($files) {
            $fileNames = ($files | ForEach-Object { $_.Name }) -join ", "
            $dispFiles = if ($fileNames.Length -gt 40) { $fileNames.Substring(0, 40) + "..." } else { $fileNames }
            Write-Host "📋 [剪貼簿變更 - 檔案] -> 數量: $($files.Count) ($dispFiles)" -ForegroundColor Magenta
        } else {
            $img = Get-Clipboard -Format Image -ErrorAction SilentlyContinue
            if ($img) {
                Write-Host "📋 [剪貼簿變更 - 圖片] -> 尺寸: $($img.Width)x$($img.Height) px" -ForegroundColor Magenta
            } else {
                $clipText = Get-Clipboard -ErrorAction SilentlyContinue
                if ($clipText) {
                    $dispClip = if ($clipText.Length -gt 40) { $clipText.Substring(0, 40) + "..." } else { $clipText }
                    Write-Host "📋 [剪貼簿變更 - 文字] -> $dispClip" -ForegroundColor Magenta
                }
            }
        }
    }

    # 4. 高效能鍵盤與滑鼠按鍵掃描 (使用 GetKeyboardState 與 GetAsyncKeyState 混合校驗)
    $null = $type::GetKeyboardState($keyStateBuffer)

    for ($vk = 1; $vk -le 254; $vk++) {
        if ($vk -eq 16 -or $vk -eq 17 -or $vk -eq 18 -or (Get-KeyName $vk) -like 'VK_*') {
            continue
        }

        # 雙重狀態確認，避免跨視窗漏抓
        $isPressed = (($keyStateBuffer[$vk] -band 0x80) -ne 0) -or (($type::GetAsyncKeyState($vk) -band 0x8000) -ne 0)

        if ($isPressed) {
            if (-not $keyStates.ContainsKey($vk)) {
                $now = [DateTime]::Now
                $keyStates[$vk] = $now
                $kName = Get-KeyName $vk
                $kColor = Get-KeyColor $vk
                $isMouse = $mouseVKs -contains $vk
                $posStr = if ($isMouse) { " " + ($type::GetMousePosString()) } else { "" }

                $activeModsCount = 0
                foreach ($mvk in $modifierVKs) {
                    if ($keyStates.ContainsKey($mvk) -and $mvk -ne $vk) {
                        $activeModsCount++
                    }
                }

                if ($activeModsCount -gt 0 -or $comboActive) {
                    if (-not $comboActive) {
                        if ($canErasePreviousSinglePress) {
                            Clear-PreviousLine
                            $canErasePreviousSinglePress = $false
                        }

                        $comboActive = $true
                        $comboChain.Clear()

                        foreach ($mvk in $modifierVKs) {
                            if ($keyStates.ContainsKey($mvk) -and $mvk -ne $vk) {
                                $comboChain.Add([pscustomobject]@{
                                    Vk = $mvk
                                    KeyName = (Get-KeyName $mvk)
                                    TimeExpr = ""
                                })
                            }
                        }

                        $delta = ($now - $lastSinglePressTime).TotalSeconds
                        $comboChain.Add([pscustomobject]@{
                            Vk = $vk
                            KeyName = $kName
                            TimeExpr = Format-Sec $delta
                        })
                        $lastComboTime = $now
                        $prevKeyReleased = $false
                    } else {
                        Clear-CurrentLine
                        $delta = ($now - $lastComboTime).TotalSeconds
                        $lastComboTime = $now

                        $timeExpr = ""
                        if ($prevKeyReleased) {
                            $timeExpr = Format-PlusSec $prevGapSec $delta
                            $prevKeyReleased = $false
                        } else {
                            $timeExpr = Format-Sec $delta
                        }

                        $comboChain.Add([pscustomobject]@{
                            Vk = $vk
                            KeyName = $kName
                            TimeExpr = $timeExpr
                        })
                    }

                    $outLine = "🔥 [組合鍵] "
                    for ($i = 0; $i -lt $comboChain.Count; $i++) {
                        $item = $comboChain[$i]
                        $itemColor = Get-KeyColor $item.Vk
                        if ($i -eq 0) {
                            $outLine += "${itemColor}$($item.KeyName)${cReset} "
                        } else {
                            $outLine += "$($item.TimeExpr) ${itemColor}$($item.KeyName)${cReset} "
                        }
                    }

                    Write-Host -NoNewline $outLine
                } else {
                    Write-Host "按下 ${kColor}${kName}${cReset}$posStr"
                    $canErasePreviousSinglePress = $true
                    $lastSinglePressVk = $vk
                    $lastSinglePressTime = $now
                }
            }
        } else {
            if ($keyStates.ContainsKey($vk)) {
                $pressTime = $keyStates[$vk]
                $duration = ([DateTime]::Now - $pressTime).TotalSeconds
                $keyStates.Remove($vk)

                $kName = Get-KeyName $vk
                $kColor = Get-KeyColor $vk
                $durStr = Format-Sec $duration
                $isMouse = $mouseVKs -contains $vk
                $posStr = if ($isMouse) { " " + ($type::GetMousePosString()) } else { "" }

                if ($comboActive) {
                    if ($modifierVKs -notcontains $vk) {
                        $prevKeyReleased = $true
                        $prevGapSec = $duration
                    }

                    $remainingMods = $false
                    foreach ($mvk in $modifierVKs) {
                        if ($keyStates.ContainsKey($mvk)) { $remainingMods = $true; break }
                    }

                    if (-not $remainingMods) {
                        Write-Host ""
                        $comboActive = $false
                        $canErasePreviousSinglePress = $false
                    }
                } else {
                    if ($canErasePreviousSinglePress -and $lastSinglePressVk -eq $vk) {
                        Clear-PreviousLine
                        Write-Host "按下又放開 ${kColor}${kName}${cReset} 持續 $durStr$posStr"
                    } else {
                        Write-Host "放開 ${kColor}${kName}${cReset} $durStr$posStr"
                    }
                    $canErasePreviousSinglePress = $false
                }
            }
        }
    }

    Start-Sleep -Milliseconds 2
}
