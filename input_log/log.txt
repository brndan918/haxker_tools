param(
    [Parameter(Mandatory=$true)]
    [int]$time
)

# 擴充 Win32 API 宣告 (包含視窗標題與輸入法/語言 API)
$signature = @"
[DllImport("user32.dll")]
public static extern short GetAsyncKeyState(int vKey);

[DllImport("user32.dll")]
public static extern IntPtr GetForegroundWindow();

[DllImport("user32.dll")]
public static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);

[DllImport("user32.dll")]
public static extern IntPtr GetKeyboardLayout(uint idThread);

[DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
public static extern int GetWindowText(IntPtr hWnd, System.Text.StringBuilder lpString, int nMaxCount);
"@

$type = Add-Type -MemberDefinition $signature -Name 'WinAPIEx' -Namespace 'User32' -PassThru
$keyStates = @{}
$startTime = [DateTime]::Now

# 全域狀態紀錄變數
$lastWindow = ""
$lastLanguage = ""
$lastClipboard = ""

function Get-KeyName([int]$vk) {
    switch ($vk) {
        1 { return '左鍵' }
        2 { return '右鍵' }
        8 { return 'Backspace' }
        9 { return 'Tab' }
        13 { return 'Enter' }
        16 { return 'Shift' }
        17 { return 'Ctrl' }
        18 { return 'Alt' }
        19 { return 'Pause' }
        20 { return 'Caps Lock' }
        27 { return 'ESC' }
        32 { return 'Space' }
        33 { return 'Page Up' }
        34 { return 'Page Down' }
        35 { return 'End' }
        36 { return 'Home' }
        37 { return '← (Left)' }
        38 { return '↑ (Up)' }
        39 { return '→ (Right)' }
        40 { return '↓ (Down)' }
        44 { return 'Print Screen' }
        45 { return 'Insert' }
        46 { return 'Delete' }
        91 { return 'Win (Left)' }
        92 { return 'Win (Right)' }
        93 { return '選單鍵 (Apps)' }
        144{ return 'Num Lock' }
        145{ return 'Scroll Lock' }
        default {
            if ($vk -ge 48 -and $vk -le 57) { return [char]$vk }
            if ($vk -ge 65 -and $vk -le 90) { return [char]$vk }
            if ($vk -ge 96 -and $vk -le 105) { return 'NumPad ' + ($vk - 96) }
            if ($vk -ge 112 -and $vk -le 123) { return 'F' + ($vk - 111) }
            switch ($vk) {
                186 { return ';' }
                187 { return '=' }
                188 { return ',' }
                189 { return '-' }
                190 { return '.' }
                191 { return '/' }
                192 { return '`' }
                219 { return '[' }
                220 { return '\' }
                221 { return ']' }
                222 { return "'" }
                default { return 'VK_' + $vk }
            }
        }
    }
}

# 取得目前焦點視窗標題與輸入法語言
function Get-ActiveContext {
    $hwnd = $type::GetForegroundWindow()
    if ($hwnd -eq [IntPtr]::Zero) { return @{ Title = "桌面/無焦點"; Lang = "未知" } }

    $sb = New-Object System.Text.StringBuilder 256
    $null = $type::GetWindowText($hwnd, $sb, $sb.Capacity)
    $title = $sb.ToString()
    if ([string]::IsNullOrWhiteSpace($title)) { $title = "無標題視窗" }

    $processId = 0
    $threadId = $type::GetWindowThreadProcessId($hwnd, [ref]$processId)
    $layout = $type::GetKeyboardLayout($threadId)
    $langId = $layout.ToInt64() -band 0xFFFF

    $langName = "未知 (0x{0:X4})" -f $langId
    try {
        $langName = (New-Object System.Globalization.CultureInfo($langId)).DisplayName
    } catch {}

    return @{ Title = $title; Lang = $langName }
}

# 檢測當前按下的修飾鍵 (Modifier keys)
function Get-ActiveModifiers {
    $mods = @()
    if (($type::GetAsyncKeyState(17) -band 0x8000) -ne 0) { $mods += "Ctrl" }
    if (($type::GetAsyncKeyState(18) -band 0x8000) -ne 0) { $mods += "Alt" }
    if (($type::GetAsyncKeyState(16) -band 0x8000) -ne 0) { $mods += "Shift" }
    if (($type::GetAsyncKeyState(91) -band 0x8000) -ne 0 -or ($type::GetAsyncKeyState(92) -band 0x8000) -ne 0) { $mods += "Win" }
    return $mods
}

# 修飾鍵 VK 列表 (避免修飾鍵自身重複偵測為組合鍵)
$modifierVKs = @(16, 17, 18, 91, 92, 160, 161, 162, 163, 164, 165)

Write-Host '====================================================' -ForegroundColor Cyan
Write-Host '  駭客風格數據監控終端（含視窗、語言、剪貼簿與組合鍵）' -ForegroundColor Cyan
if ($time -gt 0) {
    Write-Host "  倒數 $time 秒，或關閉視窗/Ctrl+C 結束" -ForegroundColor Yellow
} else {
    Write-Host '  無時間限制，請關閉視窗/Ctrl+C 結束' -ForegroundColor Yellow
}
Write-Host '====================================================' -ForegroundColor Cyan
Write-Host ''

while ($true) {
    # 1. 檢查監聽時間限制
    if ($time -gt 0) {
        $elapsed = ([DateTime]::Now - $startTime).TotalSeconds
        if ($elapsed -ge $time) {
            Write-Host ''
            Write-Host "[系統] 已達到指定的監聽時間 ($time 秒)，程式結束。" -ForegroundColor Red
            break
        }
    }

    # 2. 檢查焦點視窗與輸入語言變更
    $context = Get-ActiveContext
    if ($context.Title -ne $lastWindow -or $context.Lang -ne $lastLanguage) {
        $lastWindow = $context.Title
        $lastLanguage = $context.Lang
        Write-Host "`n[環境變更] 視窗: '$lastWindow' | 語言: $lastLanguage" -ForegroundColor Green
    }

    # 3. 檢查剪貼簿內容更新 (自動跳過重複與空內容)
    try {
        $clipText = Get-Clipboard -ErrorAction SilentlyContinue
        if ($clipText -and $clipText -ne $lastClipboard) {
            $lastClipboard = $clipText
            $displayClip = if ($clipText.Length -gt 60) { $clipText.Substring(0, 60) + "..." } else { $clipText }
            Write-Host "[剪貼簿擷取] -> $displayClip" -ForegroundColor Magenta
        }
    } catch {}

    # 4. 全鍵盤與組合鍵監聽
    for ($vk = 1; $vk -le 254; $vk++) {
        if (($vk -ge 160 -and $vk -le 165) -or (Get-KeyName $vk) -like 'VK_*') {
            continue
        }

        $state = $type::GetAsyncKeyState($vk)
        $isPressed = ($state -band 0x8000) -ne 0

        if ($isPressed) {
            if (-not $keyStates.ContainsKey($vk)) {
                $keyStates[$vk] = [DateTime]::Now
                $name = Get-KeyName $vk

                # 判斷是否觸發組合鍵
                if ($modifierVKs -notcontains $vk) {
                    $mods = Get-ActiveModifiers
                    if ($mods.Count -gt 0) {
                        $comboStr = ($mods + $name) -join ' + '
                        Write-Host " 🔥 [組合鍵] $comboStr" -ForegroundColor Yellow
                    } else {
                        Write-Host "按下 $name"
                    }
                } else {
                    Write-Host "按下 $name"
                }
            }
        } else {
            if ($keyStates.ContainsKey($vk)) {
                $pressTime = $keyStates[$vk]
                $duration = ([DateTime]::Now - $pressTime).TotalSeconds
                $keyStates.Remove($vk)
                $name = Get-KeyName $vk
                $durStr = $duration.ToString('F1')
                Write-Host "放開 $name 持續 $durStr 秒"
            }
        }
    }

    Start-Sleep -Milliseconds 15
}
